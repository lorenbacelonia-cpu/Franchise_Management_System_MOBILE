    %% ─────────────────────────────────────────────────────────────
    %% INHERITANCE: USER HIERARCHY
    %% ─────────────────────────────────────────────────────────────
    User <|-- Passenger
    User <|-- TrafficEnforcer
    User <|-- Administrator

    class User {
        +UUID id
        +String username
        +String full_name
        +String role
        +String contact_number
        +DateTime created_at
        +login(credentials) Boolean
        +logout() void
    }

    class Passenger {
        +searchFareMatrix(origin, dest) FareRate
        +scanQRCode(qrData) VehicleDetails
        +fileOverchargingComplaint(complaintData) String
        +trackComplaintStatus(ticketId) Status
    }

    class TrafficEnforcer {
        +String badge_number
        +String assigned_sector
        +verifyVehicleFranchise(plateNo) FranchiseStatus
        +selectInfractions(infractionList) List
        +calculateCumulativeFine(infractions, vehicleType) Decimal
        +issueViolationCitation(citationData) String
        +attachEvidencePhoto(imageFile) String
    }

    class Administrator {
        +String admin_passcode
        +String access_level
        +approveRenewal(mtopNo, appNo, expiryDate) void
        +updateDriverRegistry(operatorName, driverName, confirmed) void
        +adjudicateComplaint(caseId, decision) void
        +settlePenalty(caseId, amountPaid) void
        +generatePDFSummaryReport() File
        +printQRCertificate(franchiseId) File
    }
